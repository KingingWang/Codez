cmake_minimum_required(VERSION 3.20)
project(codez_runtime_host LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(codez-runtime-host main.cpp)
add_executable(codez-runtime-host-acceptance EXCLUDE_FROM_ALL acceptance.cpp)
foreach(target codez-runtime-host codez-runtime-host-acceptance)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /utf-8)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra)
    target_link_options(${target} PRIVATE -municode -static)
  endif()
endforeach()
